"""
NetSwitch - мини-приложение для переключения Windows 11 между статическим IP и DHCP.
Только стандартная библиотека Python (tkinter), без установки пакетов.
При запуске само запросит права администратора (UAC) - они нужны для netsh.
"""
import ctypes
import ipaddress
import json
import os
import subprocess
import sys
import threading
import tkinter as tk
from tkinter import ttk

APP_NAME = "NetSwitch"
CONFIG_DIR = os.path.join(os.environ.get("APPDATA", "."), APP_NAME)
CONFIG_PATH = os.path.join(CONFIG_DIR, "config.json")
NO_WINDOW = 0x08000000  # CREATE_NO_WINDOW

# ---------- цвета ----------
BG = "#14161c"
TRACK = "#1f2230"
TEXT = "#e8eaf2"
MUTED = "#7b8096"
C_STATIC = (108, 92, 231)   # фиолетовый
C_DHCP = (0, 184, 148)      # зелёный
C_BUSY = (253, 170, 60)     # янтарный
C_UNKNOWN = (58, 63, 85)    # серый
C_ERROR = (225, 85, 85)


# ---------- права администратора ----------
def ensure_admin():
    try:
        if ctypes.windll.shell32.IsUserAnAdmin():
            return
    except Exception:
        return
    exe = sys.executable
    pyw = exe.replace("python.exe", "pythonw.exe")
    if os.path.exists(pyw):
        exe = pyw
    script = os.path.abspath(sys.argv[0])
    params = " ".join([f'"{script}"'] + [f'"{a}"' for a in sys.argv[1:]])
    ctypes.windll.shell32.ShellExecuteW(None, "runas", exe, params, None, 1)
    sys.exit()


# ---------- конфиг ----------
DEFAULT_CFG = {"adapter": "", "ip": "", "mask": "255.255.255.0",
               "gateway": "", "dns1": "", "dns2": ""}


def load_cfg():
    try:
        with open(CONFIG_PATH, "r", encoding="utf-8") as f:
            return {**DEFAULT_CFG, **json.load(f)}
    except Exception:
        return dict(DEFAULT_CFG)


def save_cfg(cfg):
    os.makedirs(CONFIG_DIR, exist_ok=True)
    with open(CONFIG_PATH, "w", encoding="utf-8") as f:
        json.dump(cfg, f, ensure_ascii=False, indent=2)


# ---------- работа с сетью ----------
def run(cmd):
    p = subprocess.run(cmd, capture_output=True, creationflags=NO_WINDOW)
    out = (p.stdout + p.stderr).decode("oem", errors="replace").strip()
    return p.returncode, out


def powershell(script):
    script = "[Console]::OutputEncoding=[Text.Encoding]::UTF8; " + script
    p = subprocess.run(["powershell", "-NoProfile", "-NonInteractive", "-Command", script],
                       capture_output=True, creationflags=NO_WINDOW)
    return p.stdout.decode("utf-8", errors="replace").strip()


def list_adapters():
    out = powershell("Get-NetAdapter -Physical | Sort-Object @{e={$_.Status -ne 'Up'}} "
                     "| Select-Object -ExpandProperty Name")
    return [l.strip() for l in out.splitlines() if l.strip()]


def get_state(adapter):
    """Возвращает ('dhcp'|'static'|None, текущий IPv4)."""
    if not adapter:
        return None, ""
    a = adapter.replace("'", "''")
    out = powershell(
        f"(Get-NetIPInterface -InterfaceAlias '{a}' -AddressFamily IPv4).Dhcp; "
        f"(Get-NetIPAddress -InterfaceAlias '{a}' -AddressFamily IPv4 -ErrorAction SilentlyContinue "
        f"| Select-Object -First 1).IPAddress")
    lines = [l.strip() for l in out.splitlines() if l.strip()]
    if not lines:
        return None, ""
    mode = "dhcp" if lines[0].lower() == "enabled" else "static"
    return mode, (lines[1] if len(lines) > 1 else "")


def apply_dhcp(cfg):
    n = cfg["adapter"]
    for cmd in (["netsh", "interface", "ip", "set", "address", f"name={n}", "source=dhcp"],
                ["netsh", "interface", "ip", "set", "dns", f"name={n}", "source=dhcp"]):
        rc, out = run(cmd)
        if rc != 0:
            raise RuntimeError(out or "netsh вернул ошибку")


def apply_static(cfg):
    n = cfg["adapter"]
    cmd = ["netsh", "interface", "ip", "set", "address", f"name={n}", "static", cfg["ip"], cfg["mask"]]
    if cfg["gateway"]:
        cmd += [cfg["gateway"], "1"]
    rc, out = run(cmd)
    if rc != 0:
        raise RuntimeError(out or "netsh вернул ошибку")

    if cfg["dns1"]:
        rc, out = run(["netsh", "interface", "ip", "set", "dns", f"name={n}", "static",
                       cfg["dns1"], "primary", "validate=no"])
        if rc != 0:
            raise RuntimeError(out)
        if cfg["dns2"]:
            run(["netsh", "interface", "ip", "add", "dns", f"name={n}", cfg["dns2"],
                 "index=2", "validate=no"])
    else:
        run(["netsh", "interface", "ip", "set", "dns", f"name={n}", "source=dhcp"])


# ---------- утилиты UI ----------
def mix(c1, c2, t):
    return tuple(a + (b - a) * t for a, b in zip(c1, c2))


def hexc(c):
    return "#%02x%02x%02x" % tuple(max(0, min(255, int(v))) for v in c)


def hex2rgb(h):
    h = h.lstrip("#")
    return tuple(int(h[i:i + 2], 16) for i in (0, 2, 4))


def rr_pts(x0, y0, x1, y1, r):
    return [x0 + r, y0, x1 - r, y0, x1, y0, x1, y0 + r, x1, y1 - r, x1, y1,
            x1 - r, y1, x0 + r, y1, x0, y1, x0, y1 - r, x0, y0 + r, x0, y0]


def dark_titlebar(win):
    try:
        win.update_idletasks()
        hwnd = ctypes.windll.user32.GetParent(win.winfo_id())
        val = ctypes.c_int(1)
        ctypes.windll.dwmapi.DwmSetWindowAttribute(hwnd, 20, ctypes.byref(val), 4)
    except Exception:
        pass


# ---------- окно настроек ----------
class Settings(tk.Toplevel):
    def __init__(self, app):
        super().__init__(app.root)
        self.app = app
        self.title("Настройки")
        self.configure(bg=BG)
        self.resizable(False, False)
        self.transient(app.root)
        self.grab_set()

        cfg = app.cfg
        self.vars = {k: tk.StringVar(value=cfg[k]) for k in DEFAULT_CFG}

        frm = tk.Frame(self, bg=BG, padx=22, pady=18)
        frm.pack()

        rows = [("Адаптер", "adapter"), ("IP-адрес", "ip"), ("Маска", "mask"),
                ("Шлюз", "gateway"), ("DNS 1", "dns1"), ("DNS 2", "dns2")]
        for i, (label, key) in enumerate(rows):
            tk.Label(frm, text=label, bg=BG, fg=MUTED, font=("Segoe UI", 10),
                     anchor="w").grid(row=i, column=0, sticky="w", pady=5, padx=(0, 14))
            if key == "adapter":
                self.cb = ttk.Combobox(frm, textvariable=self.vars[key], width=26,
                                       values=app.adapters, font=("Segoe UI", 10))
                self.cb.grid(row=i, column=1, pady=5)
            else:
                tk.Entry(frm, textvariable=self.vars[key], width=28, bg=TRACK, fg=TEXT,
                         insertbackground=TEXT, relief="flat", font=("Segoe UI", 10),
                         highlightthickness=1, highlightbackground=TRACK,
                         highlightcolor=hexc(C_STATIC)).grid(row=i, column=1, pady=5, ipady=4)

        self.err = tk.Label(frm, text="", bg=BG, fg=hexc(C_ERROR), font=("Segoe UI", 9),
                            wraplength=300, justify="left")
        self.err.grid(row=len(rows), column=0, columnspan=2, sticky="w", pady=(6, 0))

        btn = tk.Label(frm, text="Сохранить", bg=hexc(C_STATIC), fg="white",
                       font=("Segoe UI", 10, "bold"), cursor="hand2", pady=8)
        btn.grid(row=len(rows) + 1, column=0, columnspan=2, sticky="ew", pady=(12, 0))
        btn.bind("<Button-1>", lambda e: self.save())
        btn.bind("<Enter>", lambda e: btn.config(bg=hexc(mix(C_STATIC, (255, 255, 255), .12))))
        btn.bind("<Leave>", lambda e: btn.config(bg=hexc(C_STATIC)))

        # позиция рядом с главным окном
        self.update_idletasks()
        x = app.root.winfo_x() + 20
        y = app.root.winfo_y() + 40
        self.geometry(f"+{x}+{y}")
        dark_titlebar(self)

    def save(self):
        v = {k: s.get().strip() for k, s in self.vars.items()}
        try:
            if not v["adapter"]:
                raise ValueError("Выберите сетевой адаптер")
            ipaddress.IPv4Address(v["ip"])
            ipaddress.IPv4Network(f"0.0.0.0/{v['mask']}")
            for k, name in (("gateway", "Шлюз"), ("dns1", "DNS 1"), ("dns2", "DNS 2")):
                if v[k]:
                    ipaddress.IPv4Address(v[k])
        except ValueError as e:
            msg = str(e)
            if "Выберите" not in msg:
                msg = "Проверьте IP, маску, шлюз и DNS - где-то некорректное значение"
            self.err.config(text=msg)
            return
        self.app.cfg = v
        save_cfg(v)
        self.app.on_settings_saved()
        self.destroy()


# ---------- главное окно ----------
class App:
    W, H = 320, 240
    TX0, TX1, TY0, TY1 = 20, 300, 74, 130   # дорожка
    PAD = 4

    def __init__(self):
        self.root = tk.Tk()
        self.root.title(APP_NAME)
        self.root.configure(bg=BG)
        self.root.resizable(False, False)
        sw, sh = self.root.winfo_screenwidth(), self.root.winfo_screenheight()
        self.root.geometry(f"{self.W}x{self.H}+{sw // 2 - self.W // 2}+{sh // 2 - self.H // 2}")

        self.cfg = load_cfg()
        self.adapters = []
        self.mode = None            # 'static' | 'dhcp' | None
        self.cur_ip = ""
        self.busy = False
        self.status = "Определяю состояние…"
        self.error = False
        self.tick_n = 0

        self.hl = 0.0               # позиция плашки 0..1 (анимируется)
        self.hl_target = 0.0
        self.col = list(C_UNKNOWN)  # текущий цвет плашки (анимируется)
        self.col_target = C_UNKNOWN

        self.c = tk.Canvas(self.root, width=self.W, height=self.H, bg=BG, highlightthickness=0)
        self.c.pack()
        self.build()
        dark_titlebar(self.root)

        threading.Thread(target=self.startup, daemon=True).start()
        self.tick()

    # --- построение интерфейса ---
    def build(self):
        c = self.c
        c.create_text(22, 28, text="NetSwitch", anchor="w", fill=TEXT, font=("Segoe UI Semibold", 15))
        self.gear = c.create_text(self.W - 24, 28, text="⚙", fill=MUTED, font=("Segoe UI Symbol", 17))
        c.tag_bind(self.gear, "<Button-1>", lambda e: self.open_settings())
        c.tag_bind(self.gear, "<Enter>", lambda e: (c.itemconfig(self.gear, fill=TEXT), c.config(cursor="hand2")))
        c.tag_bind(self.gear, "<Leave>", lambda e: (c.itemconfig(self.gear, fill=MUTED), c.config(cursor="")))

        self.track = c.create_polygon(rr_pts(self.TX0, self.TY0, self.TX1, self.TY1, 18),
                                      smooth=True, fill=TRACK, outline="")
        self.plate = c.create_polygon(rr_pts(0, 0, 10, 10, 8), smooth=True, fill=hexc(C_UNKNOWN), outline="")

        self.hw = (self.TX1 - self.TX0 - 2 * self.PAD) / 2
        cy = (self.TY0 + self.TY1) / 2
        self.lbl_static = c.create_text(self.TX0 + self.PAD + self.hw / 2, cy, text="Статика",
                                        fill=MUTED, font=("Segoe UI Semibold", 12))
        self.lbl_dhcp = c.create_text(self.TX0 + self.PAD + self.hw * 1.5, cy, text="DHCP",
                                      fill=MUTED, font=("Segoe UI Semibold", 12))

        # кликабельные зоны (поверх всего)
        for idx in (0, 1):
            x0 = self.TX0 + self.PAD + idx * self.hw
            z = c.create_rectangle(x0, self.TY0, x0 + self.hw, self.TY1, outline="", fill="")
            c.tag_bind(z, "<Button-1>", lambda e, i=idx: self.switch("static" if i == 0 else "dhcp"))
            c.tag_bind(z, "<Enter>", lambda e: c.config(cursor="hand2"))
            c.tag_bind(z, "<Leave>", lambda e: c.config(cursor=""))

        self.dot = c.create_oval(0, 0, 0, 0, fill=MUTED, outline="")
        self.status_txt = c.create_text(42, 172, text=self.status, anchor="w", fill=TEXT,
                                        font=("Segoe UI", 11), width=self.W - 60)
        self.ip_txt = c.create_text(42, 200, text="", anchor="w", fill=MUTED, font=("Segoe UI", 10))
        self.adp_txt = c.create_text(42, 222, text="", anchor="w", fill=MUTED, font=("Segoe UI", 9))

    # --- цикл анимации (~60 FPS) ---
    def tick(self):
        self.tick_n += 1
        # плавное движение плашки и цвета
        self.hl += (self.hl_target - self.hl) * 0.18
        for i in range(3):
            self.col[i] += (self.col_target[i] - self.col[i]) * 0.15

        x0 = self.TX0 + self.PAD + self.hl * self.hw
        y0, y1 = self.TY0 + self.PAD, self.TY1 - self.PAD
        self.c.coords(self.plate, *rr_pts(x0, y0, x0 + self.hw, y1, 14))
        self.c.itemconfig(self.plate, fill=hexc(self.col))

        # подсветка подписей по близости плашки
        for lbl, pos in ((self.lbl_static, 0.0), (self.lbl_dhcp, 1.0)):
            closeness = max(0.0, 1 - abs(self.hl - pos))
            self.c.itemconfig(lbl, fill=hexc(mix(hex2rgb(MUTED), (255, 255, 255), closeness)))

        # индикатор статуса (пульсирует, пока идёт применение)
        import math
        dot_col = C_ERROR if self.error else (C_BUSY if self.busy else self.col_target)
        r = 5 + (2.2 * math.sin(self.tick_n / 4) if self.busy else 0)
        self.c.coords(self.dot, 24 - r, 172 - r, 24 + r, 172 + r)
        self.c.itemconfig(self.dot, fill=hexc(dot_col))

        self.root.after(16, self.tick)

    # --- состояние ---
    def set_mode_visual(self, mode):
        self.hl_target = 1.0 if mode == "dhcp" else 0.0
        self.col_target = {"static": C_STATIC, "dhcp": C_DHCP}.get(mode, C_UNKNOWN)

    def set_status(self, text, error=False):
        self.status, self.error = text, error
        self.c.itemconfig(self.status_txt, text=text)

    def refresh_texts(self):
        self.c.itemconfig(self.ip_txt, text=f"IP: {self.cur_ip}" if self.cur_ip else "")
        self.c.itemconfig(self.adp_txt, text=f"Адаптер: {self.cfg['adapter']}" if self.cfg["adapter"] else "")

    def startup(self):
        try:
            self.adapters = list_adapters()
        except Exception:
            self.adapters = []
        if not self.cfg["adapter"] and self.adapters:
            self.cfg["adapter"] = self.adapters[0]
            save_cfg(self.cfg)
        mode, ip = get_state(self.cfg["adapter"])
        self.root.after(0, lambda: self.after_detect(mode, ip))

    def after_detect(self, mode, ip):
        self.mode, self.cur_ip = mode, ip
        self.set_mode_visual(mode)
        names = {"static": "Статический IP", "dhcp": "DHCP (автоматически)"}
        self.set_status(names.get(mode, "Адаптер не найден - откройте ⚙"), error=mode is None)
        self.refresh_texts()

    # --- действия ---
    def cfg_ready(self):
        return bool(self.cfg["adapter"] and self.cfg["ip"] and self.cfg["mask"])

    def open_settings(self):
        if not self.busy:
            Settings(self)

    def on_settings_saved(self):
        self.refresh_texts()
        self.set_status("Настройки сохранены")
        threading.Thread(target=lambda: self.root.after(
            0, lambda r=get_state(self.cfg["adapter"]): self.after_detect(*r)), daemon=True).start()

    def switch(self, target):
        if self.busy or target == self.mode:
            return
        # настройки показываем ТОЛЬКО если они ещё не заполнены
        if target == "static" and not self.cfg_ready():
            self.set_status("Сначала заполните настройки ⚙", error=True)
            self.open_settings()
            return
        if not self.cfg["adapter"]:
            self.set_status("Не выбран адаптер - откройте ⚙", error=True)
            return

        prev = self.mode
        self.busy = True
        self.error = False
        self.hl_target = 1.0 if target == "dhcp" else 0.0   # плашка едет сразу
        self.col_target = C_BUSY
        self.set_status("Применяю настройки…")

        def work():
            try:
                (apply_static if target == "static" else apply_dhcp)(self.cfg)
                import time
                time.sleep(1.2)   # даём адаптеру получить адрес
                mode, ip = get_state(self.cfg["adapter"])
                self.root.after(0, lambda: self.done_ok(target, ip))
            except Exception as e:
                self.root.after(0, lambda: self.done_fail(prev, str(e)))

        threading.Thread(target=work, daemon=True).start()

    def done_ok(self, target, ip):
        self.busy = False
        self.mode, self.cur_ip = target, ip
        self.set_mode_visual(target)
        self.set_status("Статический IP включён" if target == "static" else "DHCP включён")
        self.refresh_texts()

    def done_fail(self, prev, msg):
        self.busy = False
        self.set_mode_visual(prev)       # плашка возвращается назад
        short = (msg.splitlines()[0] if msg else "Ошибка")[:80]
        self.set_status(f"Ошибка: {short}", error=True)

    def run(self):
        self.root.mainloop()


if __name__ == "__main__":
    ensure_admin()
    App().run()
