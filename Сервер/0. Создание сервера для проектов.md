*Дальше в терминале копируем и вставляем следующие команды*
```shell
systemctl stop pveproxy

sysctl -w net.ipv6.conf.all.disable_ipv6=1 && sysctl -w net.ipv6.conf.default.disable_ipv6=1 && rm -f /etc/pve/pve-root-ca.pem /etc/pve/priv/pve-root-ca.key /etc/pve/local/pve-ssl.key /etc/pve/local/pve-ssl.pem && pvecm updatecerts -f && sysctl -w net.ipv6.conf.all.disable_ipv6=0 && sysctl -w net.ipv6.conf.default.disable_ipv6=0 && systemctl restart pvedaemon pveproxy && systemctl status pveproxy.service --no-pager -n 5

systemctl start pveproxy

reboot
```
