# -*- coding: utf-8 -*-
"""
rinput_client.py - захват клавиатуры и мыши в Windows 11 и отправка на Linux-агент.

* Только стандартная библиотека (ctypes), pip install не нужен.
* Переключение "этот ПК <-> удалённый ПК" - горячая клавиша (по умолчанию Ctrl+Alt+F12).
* В режиме "удалённый ПК" клавиатура и мышь на Windows "отключены" и всё уходит на Linux.

Запуск:
    python rinput_client.py 192.168.1.50 --secret МОЙ_ПАРОЛЬ
Опции:
    --port 24800            порт агента
    --hotkey ctrl+alt+f12   горячая клавиша переключения (ctrl/alt/shift/win + клавиша)
    --sens 1.0              множитель скорости мыши (если курсор слишком быстрый - 0.5-0.7)
"""
import argparse
import ctypes
import getpass
import hashlib
import hmac
import os
import queue
import socket
import sys
import threading
import time
import traceback
from ctypes import wintypes as wt

if sys.platform != "win32":
    sys.exit("Этот клиент работает только в Windows.")

user32 = ctypes.WinDLL("user32", use_last_error=True)
kernel32 = ctypes.WinDLL("kernel32", use_last_error=True)

LRESULT = ctypes.c_ssize_t
HOOKPROC = ctypes.WINFUNCTYPE(LRESULT, ctypes.c_int, wt.WPARAM, wt.LPARAM)

user32.SetWindowsHookExW.argtypes = [ctypes.c_int, HOOKPROC, wt.HINSTANCE, wt.DWORD]
user32.SetWindowsHookExW.restype = wt.HHOOK
user32.CallNextHookEx.argtypes = [wt.HHOOK, ctypes.c_int, wt.WPARAM, wt.LPARAM]
user32.CallNextHookEx.restype = LRESULT
user32.UnhookWindowsHookEx.argtypes = [wt.HHOOK]
user32.GetMessageW.argtypes = [ctypes.POINTER(wt.MSG), wt.HWND, wt.UINT, wt.UINT]
user32.PostThreadMessageW.argtypes = [wt.DWORD, wt.UINT, wt.WPARAM, wt.LPARAM]
user32.SetCursorPos.argtypes = [ctypes.c_int, ctypes.c_int]
kernel32.GetModuleHandleW.argtypes = [wt.LPCWSTR]
kernel32.GetModuleHandleW.restype = wt.HMODULE

WH_KEYBOARD_LL, WH_MOUSE_LL = 13, 14
WM_QUIT = 0x0012
WM_KEYDOWN, WM_KEYUP, WM_SYSKEYDOWN, WM_SYSKEYUP = 0x100, 0x101, 0x104, 0x105
WM_MOUSEMOVE = 0x200
WM_LBUTTONDOWN, WM_LBUTTONUP = 0x201, 0x202
WM_RBUTTONDOWN, WM_RBUTTONUP = 0x204, 0x205
WM_MBUTTONDOWN, WM_MBUTTONUP = 0x207, 0x208
WM_MOUSEWHEEL, WM_XBUTTONDOWN, WM_XBUTTONUP, WM_MOUSEHWHEEL = 0x20A, 0x20B, 0x20C, 0x20E
LLKHF_EXTENDED, LLKHF_INJECTED = 0x01, 0x10
LLMHF_INJECTED = 0x01


class KBDLLHOOKSTRUCT(ctypes.Structure):
    _fields_ = [("vkCode", wt.DWORD), ("scanCode", wt.DWORD), ("flags", wt.DWORD),
                ("time", wt.DWORD), ("dwExtraInfo", ctypes.c_size_t)]


class MSLLHOOKSTRUCT(ctypes.Structure):
    _fields_ = [("pt", wt.POINT), ("mouseData", wt.DWORD), ("flags", wt.DWORD),
                ("time", wt.DWORD), ("dwExtraInfo", ctypes.c_size_t)]


# ---- таблицы соответствия Windows -> Linux evdev --------------------------
# Для обычных клавиш аппаратный scan-code Windows == коду evdev (1..88).
# Расширенные (E0) клавиши и мультимедиа - по таблицам. Раскладку (EN/RU) выбирает Linux.
VK_SPECIAL = {0x13: 119, 0x03: 119,           # Pause/Break
              0x90: 69,                       # NumLock
              0x2C: 99,                       # PrintScreen (SysRq)
              0xAD: 113, 0xAE: 114, 0xAF: 115,  # mute, vol-, vol+
              0xB0: 163, 0xB1: 165, 0xB2: 166, 0xB3: 164}  # next, prev, stop, play
EXT_SCAN = {0x1C: 96, 0x1D: 97, 0x35: 98, 0x37: 99, 0x38: 100,
            0x47: 102, 0x48: 103, 0x49: 104, 0x4B: 105, 0x4D: 106,
            0x4F: 107, 0x50: 108, 0x51: 109, 0x52: 110, 0x53: 111,
            0x5B: 125, 0x5C: 126, 0x5D: 127}

BTN_LEFT, BTN_RIGHT, BTN_MIDDLE, BTN_SIDE, BTN_EXTRA = 0x110, 0x111, 0x112, 0x113, 0x114
MOUSE_BTNS = {WM_LBUTTONDOWN: (BTN_LEFT, 1), WM_LBUTTONUP: (BTN_LEFT, 0),
              WM_RBUTTONDOWN: (BTN_RIGHT, 1), WM_RBUTTONUP: (BTN_RIGHT, 0),
              WM_MBUTTONDOWN: (BTN_MIDDLE, 1), WM_MBUTTONUP: (BTN_MIDDLE, 0)}

MOD_VKS = {"ctrl": (0xA2, 0xA3), "alt": (0xA4, 0xA5), "shift": (0xA0, 0xA1), "win": (0x5B, 0x5C)}
VK_NAMES = {"pause": 0x13, "scrolllock": 0x91, "insert": 0x2D, "delete": 0x2E, "home": 0x24,
            "end": 0x23, "pgup": 0x21, "pgdn": 0x22, "esc": 0x1B, "printscreen": 0x2C}


def to_evdev(vk, scan, flags):
    if vk in VK_SPECIAL:
        return VK_SPECIAL[vk]
    if flags & LLKHF_EXTENDED:
        if scan in (0x2A, 0x36):  # "фейковый" Shift от Windows - игнорируем
            return None
        return EXT_SCAN.get(scan)
    if 1 <= scan <= 0x58:
        return scan
    return None  # в т.ч. фейковый Ctrl от AltGr (scan 0x21D)


def parse_hotkey(text):
    mods, vk = set(), None
    for t in text.lower().replace(" ", "").split("+"):
        if t in MOD_VKS:
            mods.add(t)
        elif t in VK_NAMES:
            vk = VK_NAMES[t]
        elif t.startswith("f") and t[1:].isdigit() and 1 <= int(t[1:]) <= 24:
            vk = 0x70 + int(t[1:]) - 1
        elif len(t) == 1 and t.isalnum():
            vk = ord(t.upper())
        else:
            sys.exit("Не понимаю клавишу '%s' в --hotkey" % t)
    if vk is None:
        sys.exit("В --hotkey не указана основная клавиша")
    return mods, vk


def log(msg):
    # print из хука может блокироваться (например, выделение в консоли) - выносим в поток
    threading.Thread(target=print, args=(msg,), kwargs={"flush": True}, daemon=True).start()


def encode(batch):
    out, dx, dy = [], 0, 0
    for e in batch:
        if e[0] == "m":
            dx += e[1]
            dy += e[2]
            continue
        if dx or dy:
            out.append("m %d %d\n" % (dx, dy))
            dx = dy = 0
        if e[0] == "k":
            out.append("k %d %d\n" % (e[1], e[2]))
        elif e[0] == "w":
            out.append("w %d %d\n" % (e[1], e[2]))
        elif e[0] == "r":
            out.append("r\n")
        elif e[0] == "p":
            out.append("p\n")
    if dx or dy:
        out.append("m %d %d\n" % (dx, dy))
    return "".join(out).encode()


class Link(object):
    """TCP-соединение с агентом + поток отправки + автопереподключение."""

    def __init__(self, host, port, secret):
        self.addr = (host, port)
        self.secret = secret
        self.sock = None
        self.q = queue.SimpleQueue()
        self.on_drop = None

    @property
    def connected(self):
        return self.sock is not None

    def connect(self):
        s = socket.create_connection(self.addr, timeout=5)
        try:
            s.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
            f = s.makefile("rb")
            hello = f.readline(200).split()
            if len(hello) != 2 or hello[0] != b"HELLO":
                raise ConnectionError("сервер ответил не так, как ожидалось")
            mac = hmac.new(self.secret, hello[1], hashlib.sha256).hexdigest()
            s.sendall(("AUTH %s\n" % mac).encode())
            if f.readline(20).strip() != b"OK":
                raise PermissionError("неверный пароль (--secret)")
            s.settimeout(5)
        except BaseException:
            s.close()
            raise
        self.sock = s

    def _drop(self, why):
        s, self.sock = self.sock, None
        if s:
            try:
                s.close()
            except OSError:
                pass
        if self.on_drop:
            self.on_drop()
        try:
            while True:
                self.q.get_nowait()
        except queue.Empty:
            pass
        log("[RInput] Соединение потеряно (%s). Пробую переподключиться..." % why)

    def run(self):
        last_err = None
        while True:
            if self.sock is None:
                try:
                    self.connect()
                    last_err = None
                    log("[RInput] Подключено к %s:%d" % self.addr)
                except Exception as e:
                    if str(e) != last_err:
                        last_err = str(e)
                        log("[RInput] Нет соединения: %s" % e)
                    time.sleep(2)
                    continue
            try:
                ev = self.q.get(timeout=2)
            except queue.Empty:
                ev = ("p",)  # пинг, чтобы агент видел, что мы живы
            batch = [ev]
            try:
                while True:
                    batch.append(self.q.get_nowait())
            except queue.Empty:
                pass
            try:
                self.sock.sendall(encode(batch))
            except OSError as e:
                self._drop(e)


class Capture(object):
    def __init__(self, link, hk_mods, hk_vk, sens):
        self.link, self.hk_mods, self.hk_vk, self.sens = link, hk_mods, hk_vk, sens
        self.remote = False
        self.dropped = False
        self.cx = self.cy = 0
        self.rx = self.ry = 0.0
        self.wacc = [0, 0]
        self.pressed = set()      # все физически нажатые клавиши (для горячей клавиши)
        self.local_down = set()   # нажаты, пока работали локально -> их отпускание пускаем в Windows
        self.remote_down = {}     # нажаты в удалённом режиме: vk -> evdev-код
        self.swallow = set()      # их отпускание/повтор блокируем
        self.m_local, self.m_remote = set(), set()
        self.tid = 0
        self.error = None
        link.on_drop = self.request_local

    # --- переключение режимов -------------------------------------------
    def request_local(self):  # вызывается из другого потока при обрыве связи
        self.remote = False
        self.dropped = True

    def set_local(self, notify=True):
        self.remote = False
        self.swallow.update(self.remote_down)
        self.remote_down.clear()
        self.m_remote.clear()
        if notify:
            self.link.q.put(("r",))

    def toggle(self):
        if not self.remote:
            if not self.link.connected:
                log("[RInput] Нет соединения с удалённым ПК - остаюсь в локальном режиме")
                return
            self.cx = user32.GetSystemMetrics(0) // 2
            self.cy = user32.GetSystemMetrics(1) // 2
            user32.SetCursorPos(self.cx, self.cy)
            self.rx = self.ry = 0.0
            self.wacc = [0, 0]
            self.remote = True
            log("[RInput] >>> Управление УДАЛЁННЫМ компьютером")
        else:
            self.set_local()
            log("[RInput] <<< Управление ЭТИМ компьютером")

    # --- клавиатура -------------------------------------------------------
    def _mods_ok(self):
        for m in self.hk_mods:
            a, b = MOD_VKS[m]
            if a not in self.pressed and b not in self.pressed:
                return False
        return True

    def on_key(self, vk, scan, flags, down):
        if self.dropped:
            self.dropped = False
            self.set_local(False)
        if flags & LLKHF_INJECTED:
            return False
        if down:
            self.pressed.add(vk)
            if vk in self.swallow:
                return True
            if vk == self.hk_vk and self._mods_ok():
                self.swallow.add(vk)
                self.toggle()
                return True
            if vk in self.local_down:  # автоповтор клавиши, зажатой ещё до переключения
                return self.remote
            if not self.remote:
                self.local_down.add(vk)
                return False
            if vk in self.remote_down:  # автоповтор - Linux повторяет сам
                return True
            code = to_evdev(vk, scan, flags)
            if code is not None:
                self.remote_down[vk] = code
                self.link.q.put(("k", code, 1))
            return True
        # отпускание
        self.pressed.discard(vk)
        if vk in self.swallow:
            self.swallow.discard(vk)
            return True
        if vk in self.local_down:
            self.local_down.discard(vk)
            return False
        if vk in self.remote_down:
            code = self.remote_down.pop(vk)
            if self.remote:
                self.link.q.put(("k", code, 0))
            return True
        return False

    # --- мышь ------------------------------------------------------------
    def on_mouse(self, msg, m):
        if self.dropped:
            self.dropped = False
            self.set_local(False)
        remote = self.remote
        if msg == WM_MOUSEMOVE:
            if not remote:
                return False
            dx, dy = m.pt.x - self.cx, m.pt.y - self.cy
            if dx or dy:
                self.rx += dx * self.sens
                self.ry += dy * self.sens
                ix, iy = int(self.rx), int(self.ry)
                self.rx -= ix
                self.ry -= iy
                if ix or iy:
                    self.link.q.put(("m", ix, iy))
            return True  # курсор Windows остаётся на месте (в центре экрана)

        if msg in (WM_MOUSEWHEEL, WM_MOUSEHWHEEL):
            if not remote:
                return False
            d = ctypes.c_short((m.mouseData >> 16) & 0xFFFF).value
            i = 0 if msg == WM_MOUSEWHEEL else 1
            self.wacc[i] += d
            n = int(self.wacc[i] / 120)
            self.wacc[i] -= n * 120
            if n:
                self.link.q.put(("w", n, 0) if i == 0 else ("w", 0, n))
            return True

        if msg in (WM_XBUTTONDOWN, WM_XBUTTONUP):
            code = BTN_SIDE if ((m.mouseData >> 16) & 0xFFFF) == 1 else BTN_EXTRA
            down = msg == WM_XBUTTONDOWN
        elif msg in MOUSE_BTNS:
            code, down = MOUSE_BTNS[msg]
        else:
            return False

        if down:
            if remote:
                self.m_remote.add(code)
                self.link.q.put(("k", code, 1))
                return True
            self.m_local.add(code)
            return False
        if code in self.m_local:
            self.m_local.discard(code)
            return False
        if code in self.m_remote:
            self.m_remote.discard(code)
            if remote:
                self.link.q.put(("k", code, 0))
            return True
        return False

    # --- callbacks для WinAPI -----------------------------------------------
    def _kb(self, nCode, wParam, lParam):
        try:
            if nCode == 0:
                k = KBDLLHOOKSTRUCT.from_address(lParam)
                if self.on_key(k.vkCode, k.scanCode, k.flags,
                               wParam in (WM_KEYDOWN, WM_SYSKEYDOWN)):
                    return 1
        except Exception:
            traceback.print_exc()
        return user32.CallNextHookEx(None, nCode, wParam, lParam)

    def _ms(self, nCode, wParam, lParam):
        try:
            if nCode == 0:
                m = MSLLHOOKSTRUCT.from_address(lParam)
                if not (m.flags & LLMHF_INJECTED) and self.on_mouse(wParam, m):
                    return 1
        except Exception:
            traceback.print_exc()
        return user32.CallNextHookEx(None, nCode, wParam, lParam)

    def hook_thread(self, ready):
        self.tid = kernel32.GetCurrentThreadId()
        self._kb_cb = HOOKPROC(self._kb)  # ссылки нужны, чтобы GC не удалил колбэки
        self._ms_cb = HOOKPROC(self._ms)
        hmod = kernel32.GetModuleHandleW(None)
        h1 = user32.SetWindowsHookExW(WH_KEYBOARD_LL, self._kb_cb, hmod, 0)
        h2 = user32.SetWindowsHookExW(WH_MOUSE_LL, self._ms_cb, hmod, 0)
        msg = wt.MSG()
        user32.PeekMessageW(ctypes.byref(msg), None, 0, 0, 0)  # создать очередь сообщений потока
        if not h1 or not h2:
            self.error = ctypes.WinError(ctypes.get_last_error())
            ready.set()
            return
        ready.set()
        while user32.GetMessageW(ctypes.byref(msg), None, 0, 0) > 0:
            user32.TranslateMessage(ctypes.byref(msg))
            user32.DispatchMessageW(ctypes.byref(msg))
        user32.UnhookWindowsHookEx(h1)
        user32.UnhookWindowsHookEx(h2)


def main():
    ap = argparse.ArgumentParser(description="Удалённый ввод: клиент для Windows")
    ap.add_argument("host", help="IP или имя Linux-компьютера")
    ap.add_argument("--port", type=int, default=24800)
    ap.add_argument("--secret", default=os.environ.get("RINPUT_SECRET"))
    ap.add_argument("--hotkey", default="ctrl+alt+f12")
    ap.add_argument("--sens", type=float, default=1.0)
    args = ap.parse_args()
    secret = args.secret or getpass.getpass("Пароль (secret): ")
    hk_mods, hk_vk = parse_hotkey(args.hotkey)

    user32.SetProcessDPIAware()  # координаты хука - в реальных пикселях

    link = Link(args.host, args.port, secret.encode())
    print("Подключаюсь к %s:%d ..." % (args.host, args.port))
    try:
        link.connect()
    except Exception as e:
        sys.exit("Не удалось подключиться: %s" % e)
    print("Подключено.")

    cap = Capture(link, hk_mods, hk_vk, args.sens)
    threading.Thread(target=link.run, daemon=True).start()
    ready = threading.Event()
    t = threading.Thread(target=cap.hook_thread, args=(ready,), daemon=True)
    t.start()
    ready.wait()
    if cap.error:
        sys.exit("Не удалось установить хуки: %s" % cap.error)

    print("Готово. Переключение между компьютерами: %s" % args.hotkey)
    print("Выход: Ctrl+C в этом окне (в локальном режиме).")
    try:
        while t.is_alive():
            time.sleep(0.3)
    except KeyboardInterrupt:
        pass
    finally:
        user32.PostThreadMessageW(cap.tid, WM_QUIT, 0, 0)
        link.q.put(("r",))
        time.sleep(0.3)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
rinput_agent.py - приёмник клавиатуры и мыши для Linux (ALT Workstation 10.x).

* Только стандартная библиотека. Совместим с системным python3 из ALT (3.9),
  ставить Python 3.14 на Linux НЕ нужно.
* Создаёт виртуальные клавиатуру и мышь через /dev/uinput, поэтому работает
  и в X11, и в Wayland, и даже на экране входа.
* Нужны права root (доступ к /dev/uinput).

Запуск (от root):
    python3 rinput_agent.py --secret МОЙ_ПАРОЛЬ [--port 24800]
"""
import argparse
import fcntl
import hashlib
import hmac
import os
import secrets
import signal
import socket
import struct
import sys
import time

# ---- константы linux/input.h ------------------------------------------------
EV_SYN, EV_KEY, EV_REL = 0, 1, 2
REL_X, REL_Y, REL_HWHEEL, REL_WHEEL = 0, 1, 6, 8
BTN_LEFT, BTN_EXTRA = 0x110, 0x114

# ---- ioctl-номера для /dev/uinput ----------------------------------------
# На x86/ARM/RISC-V/E2K: dir в битах 30-31. На ppc/mips/sparc - в битах 29-31.
_ALT_ARCH = os.uname().machine.startswith(("ppc", "mips", "sparc"))


def _io(nr):
    return ((1 << 29) if _ALT_ARCH else 0) | (ord("U") << 8) | nr


def _iow(nr, size):
    return ((4 << 29) if _ALT_ARCH else (1 << 30)) | (size << 16) | (ord("U") << 8) | nr


UI_SET_EVBIT = _iow(100, 4)
UI_SET_KEYBIT = _iow(101, 4)
UI_SET_RELBIT = _iow(102, 4)
UI_DEV_SETUP = _iow(3, 92)  # sizeof(struct uinput_setup) == 92
UI_DEV_CREATE = _io(1)
UI_DEV_DESTROY = _io(2)

EVENT_FMT = "llHHi"  # struct input_event (нативный размер: 24 байта на 64-бит)


class UDev(object):
    """Один виртуальный устройство uinput."""

    def __init__(self, name, keys, rels):
        fd = None
        for path in ("/dev/uinput", "/dev/input/uinput"):
            try:
                fd = os.open(path, os.O_WRONLY | os.O_NONBLOCK)
                break
            except FileNotFoundError:
                continue
            except PermissionError:
                raise SystemExit("Нет доступа к %s. Запустите агент от root (su -)." % path)
        if fd is None:
            raise SystemExit("Нет /dev/uinput. Выполните от root: modprobe uinput")
        self.fd = fd
        fcntl.ioctl(fd, UI_SET_EVBIT, EV_KEY)
        for k in keys:
            fcntl.ioctl(fd, UI_SET_KEYBIT, k)
        if rels:
            fcntl.ioctl(fd, UI_SET_EVBIT, EV_REL)
            for r in rels:
                fcntl.ioctl(fd, UI_SET_RELBIT, r)
        setup = struct.pack("HHHH80sI", 0x03, 0x1209, 0x0001, 1, name.encode(), 0)
        fcntl.ioctl(fd, UI_DEV_SETUP, setup)
        fcntl.ioctl(fd, UI_DEV_CREATE)

    def send(self, events):
        buf = b"".join(struct.pack(EVENT_FMT, 0, 0, t, c, v) for t, c, v in events)
        buf += struct.pack(EVENT_FMT, 0, 0, EV_SYN, 0, 0)
        os.write(self.fd, buf)

    def close(self):
        try:
            fcntl.ioctl(self.fd, UI_DEV_DESTROY)
        except OSError:
            pass
        os.close(self.fd)


class Injector(object):
    def __init__(self):
        self.kbd = UDev("RInput Virtual Keyboard", range(1, 256), [])
        self.mouse = UDev(
            "RInput Virtual Mouse",
            range(BTN_LEFT, BTN_EXTRA + 1),
            [REL_X, REL_Y, REL_WHEEL, REL_HWHEEL],
        )
        self.down = set()
        time.sleep(1.0)  # дать системе подхватить новые устройства

    def key(self, code, val):
        val = 1 if val else 0
        if 1 <= code <= 255:
            dev = self.kbd
        elif BTN_LEFT <= code <= BTN_EXTRA:
            dev = self.mouse
        else:
            return
        if val == (code in self.down):  # дубликат нажатия/отпускания
            return
        if val:
            self.down.add(code)
        else:
            self.down.discard(code)
        dev.send([(EV_KEY, code, val)])

    def release_all(self):
        for code in list(self.down):
            self.key(code, 0)

    def handle(self, line):
        p = line.split()
        if not p:
            return
        try:
            c = p[0]
            if c == "k":
                self.key(int(p[1]), int(p[2]))
            elif c == "m":
                dx = max(-2000, min(2000, int(p[1])))
                dy = max(-2000, min(2000, int(p[2])))
                ev = []
                if dx:
                    ev.append((EV_REL, REL_X, dx))
                if dy:
                    ev.append((EV_REL, REL_Y, dy))
                if ev:
                    self.mouse.send(ev)
            elif c == "w":
                v = max(-50, min(50, int(p[1])))
                h = max(-50, min(50, int(p[2])))
                ev = []
                if v:
                    ev.append((EV_REL, REL_WHEEL, v))
                if h:
                    ev.append((EV_REL, REL_HWHEEL, h))
                if ev:
                    self.mouse.send(ev)
            elif c == "r":
                self.release_all()
            # "p" - пинг, ничего делать не нужно
        except (ValueError, IndexError):
            pass

    def close(self):
        self.release_all()
        self.kbd.close()
        self.mouse.close()


def session(conn, addr, secret, inj):
    conn.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
    conn.settimeout(5)
    nonce = secrets.token_hex(16)
    conn.sendall(("HELLO %s\n" % nonce).encode())
    f = conn.makefile("rb")
    parts = f.readline(200).split()
    expected = hmac.new(secret, nonce.encode(), hashlib.sha256).hexdigest().encode()
    if len(parts) != 2 or parts[0] != b"AUTH" or not hmac.compare_digest(parts[1], expected):
        conn.sendall(b"DENY\n")
        print("[%s] неверный пароль, отключаю" % addr[0], flush=True)
        time.sleep(1)
        return
    conn.sendall(b"OK\n")
    print("[%s] клиент подключён" % addr[0], flush=True)
    conn.settimeout(10)  # клиент шлёт пинг каждые ~2 с
    while True:
        raw = f.readline(100)
        if not raw:
            break
        inj.handle(raw.decode("ascii", "ignore"))


def serve(bind, port, secret, inj):
    srv = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    srv.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    srv.bind((bind, port))
    srv.listen(1)
    print("Жду подключение на %s:%d ..." % (bind, port), flush=True)
    while True:
        conn, addr = srv.accept()
        try:
            session(conn, addr, secret, inj)
        except (OSError, socket.timeout) as e:
            print("[%s] соединение прервано: %s" % (addr[0], e), flush=True)
        finally:
            inj.release_all()  # на случай залипших клавиш
            try:
                conn.close()
            except OSError:
                pass
            print("[%s] отключён" % addr[0], flush=True)


def main():
    ap = argparse.ArgumentParser(description="Приёмник ввода (uinput) для Linux")
    ap.add_argument("--secret", default=os.environ.get("RINPUT_SECRET"),
                    help="общий пароль (или переменная окружения RINPUT_SECRET)")
    ap.add_argument("--port", type=int, default=24800)
    ap.add_argument("--bind", default="0.0.0.0")
    args = ap.parse_args()
    if not args.secret:
        sys.exit("Укажите --secret ПАРОЛЬ")

    signal.signal(signal.SIGTERM, lambda *a: sys.exit(0))
    inj = Injector()
    try:
        serve(args.bind, args.port, args.secret.encode(), inj)
    except KeyboardInterrupt:
        pass
    finally:
        inj.close()
        print("Остановлено.")


if __name__ == "__main__":
    main()
