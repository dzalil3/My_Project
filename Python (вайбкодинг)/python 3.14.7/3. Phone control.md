```python
"""
Phone Control — управление своим Android-телефоном с ПК (экран + полный контроль).
Работает поверх ADB и scrcpy. Только стандартная библиотека Python (tkinter).

Требования:
  1. Python 3.10+ (проверено под 3.14)
  2. adb и scrcpy в PATH (или рядом со скриптом).
     Windows: winget install Genymobile.scrcpy   (adb идёт в комплекте)
     macOS:   brew install scrcpy android-platform-tools
     Linux:   sudo apt install scrcpy adb
  3. На телефоне: Настройки -> О телефоне -> 7 раз нажать "Номер сборки"
     -> Для разработчиков -> включить "Отладка по USB".
     (на Tecno/HiOS ещё включи "Отладка по USB (настройки безопасности)" / "Install via USB",
      иначе мышью/клавиатурой управлять не получится)
"""

import os
import re
import shutil
import subprocess
import threading
import time
import tkinter as tk
from datetime import datetime
from tkinter import filedialog, messagebox, scrolledtext, ttk

NO_WINDOW = getattr(subprocess, "CREATE_NO_WINDOW", 0)


def find_tool(name: str) -> str | None:
    path = shutil.which(name)
    if path:
        return path
    here = os.path.dirname(os.path.abspath(__file__))
    for ext in ("", ".exe"):
        candidate = os.path.join(here, name + ext)
        if os.path.isfile(candidate):
            return candidate
    return None


ADB = find_tool("adb")
SCRCPY = find_tool("scrcpy")


def run_adb(*args: str, serial: str | None = None, timeout: int = 30) -> tuple[int, str]:
    if not ADB:
        return 1, "adb не найден в PATH"
    cmd = [ADB]
    if serial:
        cmd += ["-s", serial]
    cmd += list(args)
    try:
        p = subprocess.run(
            cmd, capture_output=True, text=True, timeout=timeout,
            creationflags=NO_WINDOW, encoding="utf-8", errors="replace",
        )
        return p.returncode, (p.stdout + p.stderr).strip()
    except subprocess.TimeoutExpired:
        return 1, "Таймаут команды"
    except Exception as e:  # noqa: BLE001
        return 1, str(e)


class App(tk.Tk):
    def __init__(self) -> None:
        super().__init__()
        self.title("Phone Control — Tecno Spark 40 Pro")
        self.geometry("760x640")
        self.minsize(700, 600)
        self.scrcpy_proc: subprocess.Popen | None = None
        self.devices: list[str] = []

        self._build_ui()
        self.log(f"adb:    {ADB or 'НЕ НАЙДЕН'}")
        self.log(f"scrcpy: {SCRCPY or 'НЕ НАЙДЕН'}")
        if not ADB or not SCRCPY:
            messagebox.showwarning(
                "Не хватает программ",
                "Установи scrcpy и adb (см. комментарий в начале файла) "
                "и перезапусти приложение.",
            )
        self.refresh_devices()
        self.protocol("WM_DELETE_WINDOW", self.on_close)

    # ---------------------------------------------------------------- UI
    def _build_ui(self) -> None:
        pad = {"padx": 8, "pady": 4}

        # --- устройства
        f_dev = ttk.LabelFrame(self, text="Устройство")
        f_dev.pack(fill="x", **pad)
        self.cmb_dev = ttk.Combobox(f_dev, state="readonly", width=40)
        self.cmb_dev.pack(side="left", padx=6, pady=6)
        ttk.Button(f_dev, text="Обновить", command=self.refresh_devices).pack(side="left", padx=4)
        ttk.Button(f_dev, text="Инфо", command=self.show_info).pack(side="left", padx=4)

        # --- Wi-Fi
        f_wifi = ttk.LabelFrame(self, text="Подключение по Wi-Fi")
        f_wifi.pack(fill="x", **pad)

        row1 = ttk.Frame(f_wifi)
        row1.pack(fill="x", pady=2)
        ttk.Button(row1, text="USB → Wi-Fi (авто)", command=self.usb_to_wifi).pack(side="left", padx=6)
        ttk.Label(row1, text="(подключи по USB один раз, потом кабель можно убрать)").pack(side="left")

        row2 = ttk.Frame(f_wifi)
        row2.pack(fill="x", pady=2)
        ttk.Label(row2, text="IP:порт").pack(side="left", padx=6)
        self.ent_ip = ttk.Entry(row2, width=22)
        self.ent_ip.insert(0, "192.168.0.100:5555")
        self.ent_ip.pack(side="left")
        ttk.Button(row2, text="Подключиться", command=self.wifi_connect).pack(side="left", padx=4)
        ttk.Button(row2, text="Отключить", command=self.wifi_disconnect).pack(side="left")

        row3 = ttk.Frame(f_wifi)
        row3.pack(fill="x", pady=2)
        ttk.Label(row3, text="Pair (Android 11+):").pack(side="left", padx=6)
        self.ent_pair_addr = ttk.Entry(row3, width=20)
        self.ent_pair_addr.insert(0, "IP:порт_сопряжения")
        self.ent_pair_addr.pack(side="left")
        self.ent_pair_code = ttk.Entry(row3, width=10)
        self.ent_pair_code.insert(0, "код")
        self.ent_pair_code.pack(side="left", padx=4)
        ttk.Button(row3, text="Сопряжение", command=self.wifi_pair).pack(side="left")

        # --- параметры scrcpy
        f_opt = ttk.LabelFrame(self, text="Параметры трансляции")
        f_opt.pack(fill="x", **pad)

        self.var_control = tk.BooleanVar(value=True)
        self.var_awake = tk.BooleanVar(value=True)
        self.var_screen_off = tk.BooleanVar(value=False)
        self.var_audio = tk.BooleanVar(value=True)
        self.var_top = tk.BooleanVar(value=False)
        self.var_fullscreen = tk.BooleanVar(value=False)
        self.var_record = tk.BooleanVar(value=False)
        self.var_uhid = tk.BooleanVar(value=False)

        checks = [
            ("Полное управление (мышь/клавиатура)", self.var_control),
            ("Не давать телефону засыпать", self.var_awake),
            ("Выключить экран телефона", self.var_screen_off),
            ("Звук с телефона на ПК", self.var_audio),
            ("Окно поверх всех", self.var_top),
            ("Полноэкранный режим", self.var_fullscreen),
            ("Записывать видео в файл", self.var_record),
            ("Физическая клавиатура (UHID)", self.var_uhid),
        ]
        for i, (text, var) in enumerate(checks):
            ttk.Checkbutton(f_opt, text=text, variable=var).grid(
                row=i // 2, column=i % 2, sticky="w", padx=8, pady=1)

        f_num = ttk.Frame(f_opt)
        f_num.grid(row=4, column=0, columnspan=2, sticky="w", pady=4)
        ttk.Label(f_num, text="Макс. размер (px):").pack(side="left", padx=6)
        self.ent_size = ttk.Entry(f_num, width=6)
        self.ent_size.insert(0, "1280")
        self.ent_size.pack(side="left")
        ttk.Label(f_num, text="Битрейт (Мбит/с):").pack(side="left", padx=6)
        self.ent_bitrate = ttk.Entry(f_num, width=5)
        self.ent_bitrate.insert(0, "8")
        self.ent_bitrate.pack(side="left")
        ttk.Label(f_num, text="FPS:").pack(side="left", padx=6)
        self.ent_fps = ttk.Entry(f_num, width=5)
        self.ent_fps.insert(0, "60")
        self.ent_fps.pack(side="left")

        # --- главные кнопки
        f_main = ttk.Frame(self)
        f_main.pack(fill="x", **pad)
        ttk.Button(f_main, text="▶ ЗАПУСТИТЬ ЭКРАН И УПРАВЛЕНИЕ", command=self.start_scrcpy).pack(
            side="left", expand=True, fill="x", padx=4, ipady=8)
        ttk.Button(f_main, text="■ Стоп", command=self.stop_scrcpy).pack(
            side="left", padx=4, ipady=8)

        # --- быстрые действия
        f_act = ttk.LabelFrame(self, text="Быстрые действия")
        f_act.pack(fill="x", **pad)
        actions = [
            ("Скриншот", self.screenshot),
            ("Установить APK", self.install_apk),
            ("Отправить файл", self.push_file),
            ("Забрать файл", self.pull_file),
            ("Домой", lambda: self.keyevent(3)),
            ("Назад", lambda: self.keyevent(4)),
            ("Меню задач", lambda: self.keyevent(187)),
            ("Питание", lambda: self.keyevent(26)),
            ("Громк. +", lambda: self.keyevent(24)),
            ("Громк. −", lambda: self.keyevent(25)),
            ("Перезагрузка", self.reboot),
            ("ADB-консоль", self.open_shell),
        ]
        for i, (text, cmd) in enumerate(actions):
            ttk.Button(f_act, text=text, command=cmd).grid(
                row=i // 4, column=i % 4, sticky="ew", padx=4, pady=3)
        for c in range(4):
            f_act.columnconfigure(c, weight=1)

        # --- лог
        self.txt_log = scrolledtext.ScrolledText(self, height=8, state="disabled")
        self.txt_log.pack(fill="both", expand=True, **pad)

    # ------------------------------------------------------------ helpers
    def log(self, msg: str) -> None:
        def _write() -> None:
            self.txt_log.configure(state="normal")
            self.txt_log.insert("end", f"[{datetime.now():%H:%M:%S}] {msg}\n")
            self.txt_log.see("end")
            self.txt_log.configure(state="disabled")
        self.after(0, _write)

    def bg(self, fn, *a) -> None:
        threading.Thread(target=fn, args=a, daemon=True).start()

    def serial(self) -> str | None:
        s = self.cmb_dev.get().split()[0] if self.cmb_dev.get() else None
        if not s:
            self.log("Устройство не выбрано")
        return s

    # ------------------------------------------------------------ devices
    def refresh_devices(self) -> None:
        code, out = run_adb("devices", "-l")
        self.devices = []
        labels = []
        for line in out.splitlines()[1:]:
            parts = line.split()
            if len(parts) >= 2 and parts[1] in ("device", "unauthorized", "offline"):
                model = next((p.split(":")[1] for p in parts if p.startswith("model:")), "")
                labels.append(f"{parts[0]}  {model}  [{parts[1]}]")
                self.devices.append(parts[0])
                if parts[1] == "unauthorized":
                    self.log(f"{parts[0]}: подтверди запрос отладки на экране телефона!")
        self.cmb_dev["values"] = labels
        if labels:
            self.cmb_dev.current(0)
        self.log(f"Найдено устройств: {len(labels)}")

    def show_info(self) -> None:
        s = self.serial()
        if not s:
            return

        def work() -> None:
            props = {
                "Модель": "ro.product.model",
                "Android": "ro.build.version.release",
                "Прошивка": "ro.build.display.id",
                "Производитель": "ro.product.manufacturer",
            }
            for name, prop in props.items():
                _, v = run_adb("shell", "getprop", prop, serial=s)
                self.log(f"{name}: {v}")
            _, v = run_adb("shell", "wm", "size", serial=s)
            self.log(f"Экран: {v}")
            _, v = run_adb("shell", "dumpsys", "battery", serial=s)
            m = re.search(r"level: (\d+)", v)
            if m:
                self.log(f"Батарея: {m.group(1)}%")
        self.bg(work)

    # --------------------------------------------------------------- wifi
    def usb_to_wifi(self) -> None:
        s = self.serial()
        if not s:
            return

        def work() -> None:
            _, out = run_adb("shell", "ip", "route", serial=s)
            m = re.search(r"src (\d+\.\d+\.\d+\.\d+)", out)
            if not m:
                _, out = run_adb("shell", "ip", "-f", "inet", "addr", "show", "wlan0", serial=s)
                m = re.search(r"inet (\d+\.\d+\.\d+\.\d+)", out)
            if not m:
                self.log("Не удалось определить IP телефона. Подключись к Wi-Fi.")
                return
            ip = m.group(1)
            self.log(f"IP телефона: {ip}")
            run_adb("tcpip", "5555", serial=s)
            time.sleep(2)
            _, res = run_adb("connect", f"{ip}:5555")
            self.log(res)
            self.after(0, lambda: (self.ent_ip.delete(0, "end"), self.ent_ip.insert(0, f"{ip}:5555")))
            self.after(0, self.refresh_devices)
            self.log("Готово — USB-кабель можно отключить.")
        self.bg(work)

    def wifi_connect(self) -> None:
        addr = self.ent_ip.get().strip()

        def work() -> None:
            _, res = run_adb("connect", addr)
            self.log(res)
            self.after(0, self.refresh_devices)
        self.bg(work)

    def wifi_disconnect(self) -> None:
        def work() -> None:
            _, res = run_adb("disconnect")
            self.log(res)
            self.after(0, self.refresh_devices)
        self.bg(work)

    def wifi_pair(self) -> None:
        addr = self.ent_pair_addr.get().strip()
        code = self.ent_pair_code.get().strip()

        def work() -> None:
            _, res = run_adb("pair", addr, code)
            self.log(res)
        self.bg(work)

    # ------------------------------------------------------------- scrcpy
    def start_scrcpy(self) -> None:
        if not SCRCPY:
            messagebox.showerror("Ошибка", "scrcpy не найден")
            return
        if self.scrcpy_proc and self.scrcpy_proc.poll() is None:
            self.log("scrcpy уже запущен")
            return
        s = self.serial()
        if not s:
            return

        cmd = [SCRCPY, "-s", s, "--window-title", "Мой телефон"]
        if not self.var_control.get():
            cmd.append("--no-control")
        if self.var_awake.get():
            cmd.append("--stay-awake")
        if self.var_screen_off.get():
            cmd.append("--turn-screen-off")
        if not self.var_audio.get():
            cmd.append("--no-audio")
        if self.var_top.get():
            cmd.append("--always-on-top")
        if self.var_fullscreen.get():
            cmd.append("--fullscreen")
        if self.var_uhid.get():
            cmd.append("--keyboard=uhid")
        if self.ent_size.get().strip().isdigit():
            cmd += ["--max-size", self.ent_size.get().strip()]
        if self.ent_bitrate.get().strip().isdigit():
            cmd += ["--video-bit-rate", f"{self.ent_bitrate.get().strip()}M"]
        if self.ent_fps.get().strip().isdigit():
            cmd += ["--max-fps", self.ent_fps.get().strip()]
        if self.var_record.get():
            path = filedialog.asksaveasfilename(
                defaultextension=".mp4",
                initialfile=f"phone_{datetime.now():%Y%m%d_%H%M%S}.mp4")
            if path:
                cmd += ["--record", path]

        self.log("Запуск: " + " ".join(cmd))

        def work() -> None:
            self.scrcpy_proc = subprocess.Popen(
                cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                text=True, encoding="utf-8", errors="replace",
                creationflags=NO_WINDOW)
            assert self.scrcpy_proc.stdout
            for line in self.scrcpy_proc.stdout:
                self.log(line.rstrip())
            self.log("scrcpy завершён")
        self.bg(work)

    def stop_scrcpy(self) -> None:
        if self.scrcpy_proc and self.scrcpy_proc.poll() is None:
            self.scrcpy_proc.terminate()
            self.log("scrcpy остановлен")

    # ------------------------------------------------------------ actions
    def keyevent(self, code: int) -> None:
        s = self.serial()
        if s:
            self.bg(lambda: run_adb("shell", "input", "keyevent", str(code), serial=s))

    def screenshot(self) -> None:
        s = self.serial()
        if not s:
            return
        path = filedialog.asksaveasfilename(
            defaultextension=".png",
            initialfile=f"screenshot_{datetime.now():%Y%m%d_%H%M%S}.png")
        if not path:
            return

        def work() -> None:
            run_adb("shell", "screencap", "-p", "/sdcard/_pc_shot.png", serial=s)
            _, out = run_adb("pull", "/sdcard/_pc_shot.png", path, serial=s)
            run_adb("shell", "rm", "/sdcard/_pc_shot.png", serial=s)
            self.log(out)
        self.bg(work)

    def install_apk(self) -> None:
        s = self.serial()
        if not s:
            return
        path = filedialog.askopenfilename(filetypes=[("APK", "*.apk")])
        if path:
            self.bg(lambda: self.log(run_adb("install", "-r", path, serial=s, timeout=300)[1]))

    def push_file(self) -> None:
        s = self.serial()
        if not s:
            return
        path = filedialog.askopenfilename()
        if path:
            self.bg(lambda: self.log(run_adb("push", path, "/sdcard/Download/", serial=s, timeout=600)[1]))

    def pull_file(self) -> None:
        s = self.serial()
        if not s:
            return
        remote = tk.simpledialog.askstring(
            "Забрать файл", "Путь на телефоне:", initialvalue="/sdcard/DCIM/Camera/") \
            if hasattr(tk, "simpledialog") else None
        if not remote:
            return
        folder = filedialog.askdirectory()
        if folder:
            self.bg(lambda: self.log(run_adb("pull", remote, folder, serial=s, timeout=600)[1]))

    def reboot(self) -> None:
        s = self.serial()
        if s and messagebox.askyesno("Перезагрузка", "Перезагрузить телефон?"):
            self.bg(lambda: run_adb("reboot", serial=s))

    def open_shell(self) -> None:
        s = self.serial()
        if not s or not ADB:
            return
        if os.name == "nt":
            subprocess.Popen(["cmd", "/k", ADB, "-s", s, "shell"],
                             creationflags=subprocess.CREATE_NEW_CONSOLE)
        else:
            for term in ("x-terminal-emulator", "gnome-terminal", "konsole", "xterm"):
                if shutil.which(term):
                    subprocess.Popen([term, "-e", ADB, "-s", s, "shell"])
                    return
            self.log("Терминал не найден. Выполни вручную: adb shell")

    def on_close(self) -> None:
        self.stop_scrcpy()
        self.destroy()


if __name__ == "__main__":
    import tkinter.simpledialog  # noqa: F401  (нужен для "Забрать файл")
    App().mainloop()
```
