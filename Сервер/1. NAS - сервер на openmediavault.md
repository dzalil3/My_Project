*Для удобства перед выполнением следующих пунктов настрой ssh чтобы подключаться к серверу удалённо через cmd*

*НАЧИНАЕМ НАСАТРОЙКУ ФАЙЛОВОГО ХРАНИЛИЩА*
```shell
# Создаём файл-диск на 10 ГБ
mkdir -p /srv/nas-storage
dd if=/dev/zero of=/srv/nas-storage/nas-disk.img bs=1M count=10240

# Форматируем в ext4
mkfs.ext4 /srv/nas-storage/nas-disk.img

# Создаём точку монтирования
mkdir -p /mnt/nas

# Монтируем
mount -o loop /srv/nas-storage/nas-disk.img /mnt/nas

# Добавляем в автозагрузку
echo "/srv/nas-storage/nas-disk.img /mnt/nas ext4 loop,defaults 0 0" >> /etc/fstab

# Создаём папку для файлов
mkdir -p /mnt/nas/shared
chmod 755 /mnt/nas/shared
Шаг 2: Создание пользователя net_admin
bash
# Создаём пользователя
useradd -m -s /bin/bash net_admin

# Устанавливаем пароль
echo "net_admin:P@ssw0rd" | chpasswd

# Добавляем пользователя в группу users (для доступа к общим папкам)
usermod -aG users net_admin
📁 Шаг 3: Настройка Samba (SMB) — доступ из Windows
bash
# Устанавливаем Samba
apt update && apt install samba -y

# Создаём конфигурацию
cat > /etc/samba/smb.conf << 'EOF'
[global]
   workgroup = WORKGROUP
   server string = NAS Server
   netbios name = NAS
   security = user
   map to guest = bad user
   dns proxy = no

[NAS-Storage]
   path = /mnt/nas/shared
   browseable = yes
   writeable = yes
   valid users = net_admin
   create mask = 0755
   directory mask = 0755
EOF

# Добавляем пользователя в Samba
(echo "P@ssw0rd"; echo "P@ssw0rd") | smbpasswd -a net_admin -s

# Перезапускаем Samba
systemctl restart smbd
systemctl enable smbd
🌐 Шаг 4: Настройка веб-интерфейса OMV с авторизацией
Если веб-интерфейс OMV ещё не настроен или сброшен:

bash
# Устанавливаем Nginx (если не установлен)
apt install nginx -y

# Создаём страницу входа с авторизацией
cat > /etc/nginx/sites-available/nas-web << 'EOF'
server {
    listen 80;
    server_name _;

    root /var/www/nas;
    index index.html;

    auth_basic "NAS Login";
    auth_basic_user_file /etc/nginx/.htpasswd;

    location / {
        autoindex on;
        try_files $uri $uri/ =404;
    }
}
EOF

# Создаём корневую папку для веб-интерфейса
mkdir -p /var/www/nas
ln -s /mnt/nas/shared /var/www/nas/files

# Создаём файл с паролями для Nginx
apt install apache2-utils -y
htpasswd -cb /etc/nginx/.htpasswd net_admin P@ssw0rd

# Активируем сайт
ln -sf /etc/nginx/sites-available/nas-web /etc/nginx/sites-enabled/
rm -f /etc/nginx/sites-enabled/default

systemctl restart nginx
systemctl enable --now nginx

# Меняем владельца папки на net_admin и группу users
chown -R net_admin:users /mnt/nas/shared

# Даём права на чтение и запись для владельца и группы
chmod -R 775 /mnt/nas/shared
```

Перепишем `smb.conf` в чистом виде (без `[homes]` и `[printers]`)
```
vim /etc/samba/smb.conf
```
Удаляем всё содержимое и вставляем следующий конфиг:
```
[global]
   workgroup = WORKGROUP
   server string = NAS Server
   netbios name = NAS
   security = user
   map to guest = bad user
   dns proxy = no
   
   # Протоколы и безопасность
   min protocol = SMB2
   max protocol = SMB3
   server signing = auto

   # Отключаем лишние службы печати, чтобы Windows не падала в ошибку
   load printers = no
   printing = bsd
   printcap name = /dev/null
   disable spoolss = yes

[NAS-Storage]
   path = /mnt/nas/shared
   browseable = yes
   writeable = yes
   valid users = net_admin
   create mask = 0755
   directory mask = 0755
   force user = net_admin
```

```
sudo systemctl restart smbd nmbd
```
Создаем пользователя и группу на сервере:
```
groupadd -f users

useradd -m -s /bin/bash -g users net_admin

echo "net_admin:P@ssw0rd" | chpasswd

(echo "P@ssw0rd"; echo "P@ssw0rd") | smbpasswd -a net_admin -s
```
Выдача прав:
```
chown -R net_admin:users /mnt/nas/shared

find /mnt/nas/shared -type d -exec chmod 775 {} +
find /mnt/nas/shared -type f -exec chmod 664 {} +

chmod 755 /mnt/nas
```
