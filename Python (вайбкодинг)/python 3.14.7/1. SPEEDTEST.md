```python
"""
Анимированный спидометр для измерения скорости интернета.

Требования:
    pip install speedtest-cli

Запуск:
    python speedometr.py

Сборка в .exe:
    python -m PyInstaller --onefile --windowed --name Speedometer --hidden-import speedtest speedometr.py

Работает на Python 3.10+ (в т.ч. на 3.14.x).
"""

import math
import random
import subprocess
import sys
import threading
import tkinter as tk
from dataclasses import dataclass

import os
import builtins

# ---------------------------------------------------------------------------
#  Фиксы для работы speedtest-cli внутри .exe (PyInstaller --windowed)
# ---------------------------------------------------------------------------
# 1) В windowed-режиме нет консоли, поэтому sys.stdout и sys.stderr равны None.
#    speedtest-cli при импорте вызывает sys.stdout.fileno() и падает.
#    Подменяем их на /dev/null — объект с рабочим fileno().
if sys.stdout is None:
    sys.stdout = open(os.devnull, "w")
if sys.stderr is None:
    sys.stderr = open(os.devnull, "w")

# 2) speedtest-cli написан под Python 2 и делает "import __builtin__".
#    В Python 3 модуль называется builtins — регистрируем его под старым именем.
sys.modules["__builtin__"] = builtins
# ---------------------------------------------------------------------------


def _ensure_speedtest_cli():
    """Автоматически ставит speedtest-cli, если его нет.

    Работает только когда скрипт запущен как обычный .py через интерпретатор
    Python (есть pip). Если программа уже собрана в .exe через PyInstaller,
    библиотека вшита внутрь заранее и эта функция даже не вызывается —
    sys.frozen будет True.
    """
    if getattr(sys, "frozen", False):
        return  # это .exe — зависимость уже упакована внутри
    try:
        import speedtest  # noqa: F401
        return
    except ImportError:
        pass

    try:
        subprocess.check_call(
            [sys.executable, "-m", "pip", "install", "--quiet", "speedtest-cli"]
        )
    except Exception:
        pass  # не получилось — просто останемся без модуля, обработаем ниже


_ensure_speedtest_cli()

try:
    import speedtest
except ImportError:
    speedtest = None


# ---------------------------------------------------------------------------
#  Настройки внешнего вида
# ---------------------------------------------------------------------------

WIDTH, HEIGHT = 720, 640
BG_TOP = "#060a16"
BG_BOTTOM = "#0c1730"
ACCENT = "#39c9ff"       # основной сдержанный акцент (голубой)
ACCENT_DIM = "#3a4a72"   # приглушённый сине-серый для сетки/частиц
GRID_COLOR = "#16223f"   # линии тех-сетки, почти сливаются с фоном

GAUGE_CX, GAUGE_CY, GAUGE_R = WIDTH // 2, 330, 220
START_ANGLE = 225        # градусы, начало шкалы (0 Mbps)
SWEEP_ANGLE = -270       # шкала идёт по часовой стрелке на 270°
MAX_SPEED_DEFAULT = 100  # шкала будет растягиваться, если скорость больше


def lerp(a, b, t):
    return a + (b - a) * t


def ease_out(t):
    return 1 - (1 - t) ** 3


# ---------------------------------------------------------------------------
#  Сдержанный тех-фон: сетка, редкие частицы, сигнальные импульсы
# ---------------------------------------------------------------------------

@dataclass
class Particle:
    x: float
    y: float
    r: float
    speed: float
    alpha_phase: float


@dataclass
class Pulse:
    radius: float = 0.0
    max_radius: float = 260.0
    speed: float = 1.6


class AnimatedBackground:
    """Градиентный фон + медленная тех-сетка + редкие частицы + сигнальные импульсы.

    Никакого "конфетти": палитра ограничена оттенками синего/голубого,
    движение медленное и равномерное — задача создать ощущение работающей
    сети, а не праздничную анимацию.
    """

    GRID_STEP = 46
    PARTICLE_COUNT = 16
    PULSE_INTERVAL = 130   # кадров между новыми импульсами

    def __init__(self, canvas: tk.Canvas):
        self.canvas = canvas
        self.particles = [self._new_particle(random_y=True) for _ in range(self.PARTICLE_COUNT)]
        self.pulses: list[Pulse] = []
        self._grid_offset = 0.0
        self._frame = 0
        self._draw_gradient()

    def _new_particle(self, random_y=False) -> Particle:
        return Particle(
            x=random.uniform(0, WIDTH),
            y=random.uniform(0, HEIGHT) if random_y else HEIGHT + 10,
            r=random.uniform(1.0, 2.0),
            speed=random.uniform(0.15, 0.45),
            alpha_phase=random.uniform(0, math.pi * 2),
        )

    def _draw_gradient(self):
        steps = 40
        for i in range(steps):
            t = i / steps
            color = self._blend(BG_TOP, BG_BOTTOM, t)
            y0 = HEIGHT * t
            y1 = HEIGHT * (t + 1 / steps)
            self.canvas.create_rectangle(0, y0, WIDTH, y1, fill=color,
                                          outline=color, tags="bg_gradient")

    @staticmethod
    def _blend(c1, c2, t):
        c1 = c1.lstrip("#")
        c2 = c2.lstrip("#")
        r1, g1, b1 = int(c1[0:2], 16), int(c1[2:4], 16), int(c1[4:6], 16)
        r2, g2, b2 = int(c2[0:2], 16), int(c2[2:4], 16), int(c2[4:6], 16)
        r = int(lerp(r1, r2, t))
        g = int(lerp(g1, g2, t))
        b = int(lerp(b1, b2, t))
        return f"#{r:02x}{g:02x}{b:02x}"

    def _bg_color_at(self, y):
        t = max(0.0, min(1.0, y / HEIGHT))
        return self._blend(BG_TOP, BG_BOTTOM, t)

    # -- слои -----------------------------------------------------------
    def _draw_grid(self):
        self.canvas.delete("grid")
        self._grid_offset = (self._grid_offset + 0.4) % self.GRID_STEP
        y = -self.GRID_STEP + self._grid_offset
        while y < HEIGHT:
            # линии слегка растворяются в фоне сверху/снизу — без резкости
            local_bg = self._bg_color_at(y)
            color = self._blend(local_bg, GRID_COLOR, 0.9)
            self.canvas.create_line(0, y, WIDTH, y, fill=color, width=1, tags="grid")
            y += self.GRID_STEP
        for x in range(0, WIDTH, self.GRID_STEP):
            local_bg = self._bg_color_at(HEIGHT / 2)
            color = self._blend(local_bg, GRID_COLOR, 0.9)
            self.canvas.create_line(x, 0, x, HEIGHT, fill=color, width=1, tags="grid")
        self.canvas.tag_lower("grid")
        self.canvas.tag_raise("grid", "bg_gradient")

    def _draw_particles(self):
        self.canvas.delete("particle")
        for p in self.particles:
            p.y -= p.speed
            p.alpha_phase += 0.03
            if p.y < -10:
                new = self._new_particle(random_y=False)
                p.x, p.y, p.r, p.speed = new.x, new.y, new.r, new.speed

            pulse = 0.4 + 0.4 * math.sin(p.alpha_phase)
            local_bg = self._bg_color_at(p.y)
            color = self._blend(local_bg, ACCENT_DIM, 0.55 + 0.25 * pulse)
            self.canvas.create_oval(
                p.x - p.r, p.y - p.r, p.x + p.r, p.y + p.r,
                fill=color, outline="", tags="particle"
            )

    def _draw_pulses(self):
        self.canvas.delete("pulse")
        self._frame += 1
        if self._frame % self.PULSE_INTERVAL == 0:
            self.pulses.append(Pulse())

        alive = []
        for pulse in self.pulses:
            pulse.radius += pulse.speed
            t = pulse.radius / pulse.max_radius
            if t >= 1.0:
                continue
            fade = 1.0 - t  # чем шире кольцо, тем бледнее
            local_bg = self._bg_color_at(GAUGE_CY)
            color = self._blend(local_bg, ACCENT, 0.12 + 0.18 * fade)
            self.canvas.create_oval(
                GAUGE_CX - pulse.radius, GAUGE_CY - pulse.radius,
                GAUGE_CX + pulse.radius, GAUGE_CY + pulse.radius,
                outline=color, width=1, tags="pulse"
            )
            alive.append(pulse)
        self.pulses = alive

    def step(self):
        self._draw_grid()
        self._draw_pulses()
        self._draw_particles()


# ---------------------------------------------------------------------------
#  Спидометр
# ---------------------------------------------------------------------------

class Speedometer:
    def __init__(self, canvas: tk.Canvas):
        self.canvas = canvas
        self.max_speed = MAX_SPEED_DEFAULT
        self.current_value = 0.0    # то, что реально отображает стрелка
        self.target_value = 0.0     # к чему стрелка стремится
        self._glow_phase = 0.0

    def set_target(self, value, max_speed=None):
        if max_speed:
            self.max_speed = max_speed
        self.target_value = max(0, value)

    def value_to_angle(self, value):
        t = min(value / self.max_speed, 1.0) if self.max_speed else 0
        return START_ANGLE + SWEEP_ANGLE * t

    def draw(self):
        c = self.canvas
        c.delete("gauge")

        # --- Внешнее неоновое кольцо со свечением (несколько наложенных дуг) ---
        self._glow_phase += 0.05
        glow_pulse = 0.5 + 0.5 * math.sin(self._glow_phase)
        for i, width in enumerate([18, 12, 6]):
            alpha_color = "#161c4a"
            c.create_arc(
                GAUGE_CX - GAUGE_R - 10, GAUGE_CY - GAUGE_R - 10,
                GAUGE_CX + GAUGE_R + 10, GAUGE_CY + GAUGE_R + 10,
                start=START_ANGLE, extent=SWEEP_ANGLE,
                style="arc", outline=alpha_color, width=width + i * 2,
                tags="gauge"
            )

        # --- Цветные зоны шкалы: зелёная / жёлтая / красная ---
        zones = [
            (0.0, 0.5, "#25d366"),
            (0.5, 0.8, "#ffcc00"),
            (0.8, 1.0, "#ff3b3b"),
        ]
        for t0, t1, color in zones:
            a0 = START_ANGLE + SWEEP_ANGLE * t0
            extent = SWEEP_ANGLE * (t1 - t0)
            c.create_arc(
                GAUGE_CX - GAUGE_R, GAUGE_CY - GAUGE_R,
                GAUGE_CX + GAUGE_R, GAUGE_CY + GAUGE_R,
                start=a0, extent=extent, style="arc",
                outline=color, width=14, tags="gauge"
            )

        # --- Деления и подписи ---
        divisions = 10
        for i in range(divisions + 1):
            t = i / divisions
            angle = START_ANGLE + SWEEP_ANGLE * t
            rad = math.radians(angle)
            x0 = GAUGE_CX + (GAUGE_R - 22) * math.cos(rad)
            y0 = GAUGE_CY - (GAUGE_R - 22) * math.sin(rad)
            x1 = GAUGE_CX + (GAUGE_R - 4) * math.cos(rad)
            y1 = GAUGE_CY - (GAUGE_R - 4) * math.sin(rad)
            c.create_line(x0, y0, x1, y1, fill="#e8e8ff", width=2, tags="gauge")

            lx = GAUGE_CX + (GAUGE_R - 42) * math.cos(rad)
            ly = GAUGE_CY - (GAUGE_R - 42) * math.sin(rad)
            label = round(self.max_speed * t)
            c.create_text(lx, ly, text=str(label), fill="#9aa0d6",
                          font=("Consolas", 10, "bold"), tags="gauge")

        # --- Плавное движение стрелки к target_value ---
        self.current_value = lerp(self.current_value, self.target_value, 0.08)
        angle = self.value_to_angle(self.current_value)
        rad = math.radians(angle)

        needle_len = GAUGE_R - 30
        tip_x = GAUGE_CX + needle_len * math.cos(rad)
        tip_y = GAUGE_CY - needle_len * math.sin(rad)

        # свечение стрелки (толстая полупрозрачная линия под основной)
        c.create_line(GAUGE_CX, GAUGE_CY, tip_x, tip_y,
                      fill="#39c9ff", width=8, capstyle="round",
                      stipple="gray25", tags="gauge")
        c.create_line(GAUGE_CX, GAUGE_CY, tip_x, tip_y,
                      fill="#ffffff", width=3, capstyle="round", tags="gauge")

        c.create_oval(GAUGE_CX - 12, GAUGE_CY - 12, GAUGE_CX + 12, GAUGE_CY + 12,
                      fill="#39c9ff", outline="#ffffff", width=2, tags="gauge")

        # --- Числовое значение по центру снизу дуги ---
        c.create_text(
            GAUGE_CX, GAUGE_CY + 70,
            text=f"{self.current_value:5.1f}",
            fill="#ffffff", font=("Consolas", 40, "bold"), tags="gauge"
        )
        c.create_text(
            GAUGE_CX, GAUGE_CY + 110,
            text="Mbps", fill="#9aa0d6",
            font=("Consolas", 16, "bold"), tags="gauge"
        )


# ---------------------------------------------------------------------------
#  Основное приложение
# ---------------------------------------------------------------------------

class SpeedTestApp:
    def __init__(self, root: tk.Tk):
        self.root = root
        self.root.title("Speedometer — тест скорости интернета")
        self.root.resizable(False, False)

        self.canvas = tk.Canvas(root, width=WIDTH, height=HEIGHT,
                                 highlightthickness=0, bg=BG_TOP)
        self.canvas.pack()

        self.background = AnimatedBackground(self.canvas)
        self.gauge = Speedometer(self.canvas)

        self.status_text_id = self.canvas.create_text(
            WIDTH // 2, 60, text="Готов к тесту",
            fill="#ffffff", font=("Consolas", 20, "bold")
        )

        self.info_text_id = self.canvas.create_text(
            WIDTH // 2, HEIGHT - 90,
            text="Ping: -- ms      Download: -- Mbps      Upload: -- Mbps",
            fill="#c8c8ff", font=("Consolas", 13)
        )

        self.button = tk.Button(
            root, text="▶ Начать тест", command=self.start_test,
            font=("Consolas", 14, "bold"), bg=ACCENT, fg="#060a16",
            activebackground="#ff00e6", activeforeground="#ffffff",
            relief="flat", padx=20, pady=10, cursor="hand2"
        )
        self.button_window = self.canvas.create_window(
            WIDTH // 2, HEIGHT - 35, window=self.button
        )

        self._testing = False
        self._last_ping = None
        self._last_download = None
        self._last_upload = None

        self._animate()

    # -- анимационный цикл (фон + стрелка), ~50 fps ------------------------
    def _animate(self):
        self.background.step()
        self.gauge.draw()
        self.canvas.tag_raise(self.status_text_id)
        self.canvas.tag_raise(self.info_text_id)
        self.root.after(20, self._animate)

    # -- запуск теста в отдельном потоке, чтобы GUI не подвисал ------------
    def start_test(self):
        if self._testing:
            return
        if speedtest is None:
            self.canvas.itemconfigure(
                self.status_text_id,
                text="Установите библиотеку: pip install speedtest-cli"
            )
            return

        self._testing = True
        self.button.config(state="disabled", text="Идёт тест...")
        self.gauge.set_target(0)
        self.canvas.itemconfigure(self.status_text_id, text="Подключение к серверу...")

        thread = threading.Thread(target=self._run_speedtest, daemon=True)
        thread.start()

    def _run_speedtest(self):
        try:
            st = speedtest.Speedtest()
            st.get_best_server()

            self._set_status("Измеряем Ping...")
            ping = st.results.ping

            self._set_status("Измеряем скорость загрузки...")
            download_bps = st.download(threads=None,
                                        callback=self._download_callback)
            download_mbps = download_bps / 1_000_000

            self._set_status("Измеряем скорость отдачи...")
            upload_bps = st.upload(threads=None,
                                    callback=self._upload_callback)
            upload_mbps = upload_bps / 1_000_000

            self.root.after(0, self._finish_test, ping, download_mbps, upload_mbps)
        except Exception as exc:
            self.root.after(0, self._on_error, str(exc))

    # -- колбэки speedtest во время измерения (вызываются из потока) -------
    def _download_callback(self, current, total, start=False, end=False):
        mbps = (current * 8) / 1_000_000 if current else 0
        self.root.after(0, self._update_live_value, mbps, "Download")

    def _upload_callback(self, current, total, start=False, end=False):
        mbps = (current * 8) / 1_000_000 if current else 0
        self.root.after(0, self._update_live_value, mbps, "Upload")

    def _update_live_value(self, mbps, label):
        max_speed = max(MAX_SPEED_DEFAULT, mbps * 1.3)
        self.gauge.set_target(mbps, max_speed=max_speed)
        self.canvas.itemconfigure(
            self.status_text_id, text=f"{label}: измерение..."
        )

    def _set_status(self, text):
        self.root.after(0, lambda: self.canvas.itemconfigure(
            self.status_text_id, text=text))

    def _finish_test(self, ping, download_mbps, upload_mbps):
        self._last_ping = ping
        self._last_download = download_mbps
        self._last_upload = upload_mbps

        self.canvas.itemconfigure(
            self.status_text_id, text="Тест завершён ✔"
        )
        self.canvas.itemconfigure(
            self.info_text_id,
            text=(f"Ping: {ping:.0f} ms      "
                  f"Download: {download_mbps:.1f} Mbps      "
                  f"Upload: {upload_mbps:.1f} Mbps")
        )
        self.gauge.set_target(download_mbps, max_speed=max(MAX_SPEED_DEFAULT, download_mbps * 1.3))

        self.button.config(state="normal", text="▶ Повторить тест")
        self._testing = False

    def _on_error(self, message):
        self.canvas.itemconfigure(
            self.status_text_id, text=f"Ошибка: {message}"
        )
        self.button.config(state="normal", text="▶ Начать тест")
        self._testing = False


def main():
    root = tk.Tk()
    app = SpeedTestApp(root)
    root.mainloop()


if __name__ == "__main__":
    main()
```
